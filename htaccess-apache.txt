# .htaccess para agustincortesmarcos.com (Apache o LiteSpeed, hosting cPanel)
#
# Cómo instalarlo: copia este contenido en el archivo .htaccess de public_html.
# Si ya existe uno (cPanel suele añadir bloques como «php -- BEGIN cPanel-generated
# handler»), NO lo sustituyas: guarda antes una copia y pega esto debajo de lo que haya.
# Después comprueba que la web carga. Si aparece un error 500, restaura la copia.

Options -MultiViews
DirectoryIndex home.html index.html

<IfModule mod_rewrite.c>
RewriteEngine On

# 1. Un solo dominio canónico: https://agustincortesmarcos.com (sin www).
#    Hoy www.agustincortesmarcos.com sirve la misma web sin redirigir: contenido duplicado.
RewriteCond %{HTTP_HOST} ^www\.agustincortesmarcos\.com$ [NC]
RewriteRule ^ https://agustincortesmarcos.com%{REQUEST_URI} [R=301,L]

RewriteCond %{HTTPS} !=on
RewriteCond %{HTTP:X-Forwarded-Proto} !=https
RewriteRule ^ https://agustincortesmarcos.com%{REQUEST_URI} [R=301,L]

# 2. Copias antiguas de las fichas subidas por error dentro de /assets/.
RewriteRule ^assets/(cuentos-de-la-abuela-robot|el-ano-que-solo-hubo-primavera|intemperie|la-pesadilla-de-bradbury|las-reglas-de-la-casa|nadie-nace-sabiendo)(?:/.*)?$ https://agustincortesmarcos.com/libros/$1/ [R=301,L,NC]

# 3. /home.html, /en/home.html y .../index.html → URL de carpeta.
#    Solo mira la petición original (THE_REQUEST), así no choca con DirectoryIndex.
RewriteCond %{THE_REQUEST} ^[A-Z]{3,9}\s/+((?:[^?\s]*/)?)(?:home|index)\.html[?\s] [NC]
RewriteRule ^ https://agustincortesmarcos.com/%1 [R=301,L]

# 4. sitemap.xml y robots.txt se sirven tal cual.
RewriteRule ^(?:sitemap\.xml|robots\.txt)$ - [L]
</IfModule>

<IfModule mod_mime.c>
AddType application/xml .xml
AddType text/plain .txt
AddType image/webp .webp
AddType image/svg+xml .svg
</IfModule>

<FilesMatch "^(sitemap\.xml|robots\.txt)$">
  Require all granted
</FilesMatch>

# 5. Compresión de texto (HTML, CSS, JS, SVG, XML): páginas más ligeras.
<IfModule mod_deflate.c>
AddOutputFilterByType DEFLATE text/html text/css text/plain text/xml application/xml application/javascript text/javascript application/json application/ld+json image/svg+xml
</IfModule>

# 6. Caché de imágenes en el navegador del visitante (7 días).
#    No se aplica a HTML, CSS ni JS para que los cambios que subas se vean al momento.
<IfModule mod_expires.c>
ExpiresActive On
ExpiresByType image/webp "access plus 7 days"
ExpiresByType image/jpeg "access plus 7 days"
ExpiresByType image/png "access plus 7 days"
ExpiresByType image/svg+xml "access plus 7 days"
</IfModule>
